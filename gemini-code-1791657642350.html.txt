    <script>
        window.onload = function() {
            lucide.createIcons();
            checkAuthSession();
        };

        function checkAuthSession() {
            const isLogged = sessionStorage.getItem('admin_logged_in');
            if (isLogged === 'true') {
                document.getElementById('login-container').classList.add('hidden');
                document.getElementById('dashboard-container').classList.remove('hidden');
                loadDashboardData();
            } else {
                document.getElementById('login-container').classList.remove('hidden');
                document.getElementById('dashboard-container').classList.add('hidden');
            }
        }

        function handleLogin(e) {
            e.preventDefault();
            const user = document.getElementById('login-username').value.trim();
            const pass = document.getElementById('login-password').value.trim();

            if (user === 'admin' && pass === '1109') {
                sessionStorage.setItem('admin_logged_in', 'true');
                checkAuthSession();
                showToast('Амжилттай нэвтэрлээ!');
            } else {
                showToast('Админ нэр эсвэл нууц үг буруу байна!');
            }
        }

        function handleLogout() {
            sessionStorage.removeItem('admin_logged_in');
            document.getElementById('login-username').value = '';
            document.getElementById('login-password').value = '';
            checkAuthSession();
            showToast('Системээс гарлаа.');
        }

        function loadDashboardData() {
            loadStats();
            loadOrders();
            loadVisitors();
        }

        function loadStats() {
            const totalVisits = localStorage.getItem('total_visits') || '0';
            document.getElementById('stat-total-visits').innerText = parseInt(totalVisits).toLocaleString();

            let orders = getAllOrders();
            document.getElementById('stat-total-orders').innerText = orders.length;

            let pendingCount = orders.filter(o => o.status === 'Хүлээгдэж байна').length;
            document.getElementById('stat-pending-orders').innerText = pendingCount;

            let totalRev = orders.reduce((acc, curr) => {
                let num = parseInt(curr.total.replace(/[^0-9]/g, '')) || 0;
                return acc + num;
            }, 0);
            document.getElementById('stat-total-revenue').innerText = totalRev.toLocaleString() + ' ₮';
        }

        function getAllOrders() {
            let orders = [];
            for (let i = 0; i < localStorage.length; i++) {
                let key = localStorage.key(i);
                if (key && key.startsWith('order_')) {
                    try {
                        let data = JSON.parse(localStorage.getItem(key));
                        orders.push(data);
                    } catch(e) {}
                }
            }
            return orders;
        }

        function loadOrders() {
            let orders = getAllOrders();
            let tbody = document.getElementById('orders-table-body');
            tbody.innerHTML = '';

            if (orders.length === 0) {
                tbody.innerHTML = `<tr><td colspan="7" class="p-6 text-center text-zinc-500 text-xs">Захиалга байхгүй байна.</td></tr>`;
                return;
            }

            orders.forEach((ord) => {
                let tr = document.createElement('tr');
                tr.className = 'hover:bg-zinc-900/40 transition';
                tr.innerHTML = `
                    <td class="p-3.5 font-bold text-indigo-400">#${ord.code}</td>
                    <td class="p-3.5">
                        <span class="block font-semibold text-white uppercase text-xs">${ord.platform}</span>
                        <span class="text-[11px] text-zinc-400">${ord.service} (${ord.quantity})</span>
                    </td>
                    <td class="p-3.5 text-zinc-300 max-w-xs truncate"><a href="${ord.target.startsWith('http') ? ord.target : '#'}" target="_blank" class="hover:underline text-indigo-300">${ord.target}</a></td>
                    <td class="p-3.5 text-zinc-300">${ord.contact}</td>
                    <td class="p-3.5 font-bold text-white">${ord.total}</td>
                    <td class="p-3.5">
                        <span class="px-2.5 py-1 rounded-full text-[10px] font-bold ${ord.status === 'Хүлээгдэж байна' ? 'bg-amber-500/10 text-amber-400 border border-amber-500/20' : 'bg-emerald-500/10 text-emerald-400 border border-emerald-500/20'}">${ord.status}</span>
                    </td>
                    <td class="p-3.5 text-right space-x-2">
                        <button onclick="toggleOrderStatus('${ord.code}')" class="px-2.5 py-1 rounded-lg bg-zinc-800 hover:bg-zinc-700 text-zinc-300 text-xs transition">Төлөв солих</button>
                        <button onclick="deleteOrder('${ord.code}')" class="px-2.5 py-1 rounded-lg bg-rose-500/10 hover:bg-rose-500/20 text-rose-400 text-xs transition">Устгах</button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        function toggleOrderStatus(code) {
            let key = 'order_' + code;
            let data = JSON.parse(localStorage.getItem(key));
            if (data) {
                data.status = data.status === 'Хүлээгдэж байна' ? 'Биелэгдсэн (Дууссан)' : 'Хүлээгдэж байна';
                localStorage.setItem(key, JSON.stringify(data));
                loadDashboardData();
                showToast(`Захиалга #${code} төлөв шинэчлэгдлээ.`);
            }
        }

        function deleteOrder(code) {
            if (confirm(`Захиалга #${code}-ийг устгахдаа итгэлтэй байна уу?`)) {
                localStorage.removeItem('order_' + code);
                loadDashboardData();
                showToast(`Захиалга #${code} устгагдлаа.`);
            }
        }

        function loadVisitors() {
            let visitors = JSON.parse(localStorage.getItem('visit_logs') || '[]');
            let tbody = document.getElementById('visitors-table-body');
            tbody.innerHTML = '';

            if (visitors.length === 0) {
                tbody.innerHTML = `<tr><td colspan="3" class="p-6 text-center text-zinc-500 text-xs">Зочилсон бүртгэл байхгүй байна.</td></tr>`;
                return;
            }

            visitors.forEach((v, index) => {
                let tr = document.createElement('tr');
                tr.className = 'hover:bg-zinc-900/40 transition';
                tr.innerHTML = `
                    <td class="p-3.5 text-zinc-500 font-mono text-xs">${index + 1}</td>
                    <td class="p-3.5 text-zinc-300 font-medium">${v.time}</td>
                    <td class="p-3.5 text-zinc-400 text-xs">${v.ua}</td>
                `;
                tbody.appendChild(tr);
            });
        }

        function clearAllData() {
            if (confirm('Бүх захиалга болон зочдын бүртгэлийг устгах уу?')) {
                for (let i = localStorage.length - 1; i >= 0; i--) {
                    let key = localStorage.key(i);
                    if (key && (key.startsWith('order_') || key === 'visit_logs' || key === 'total_visits')) {
                        localStorage.removeItem(key);
                    }
                }
                loadDashboardData();
                showToast('Бүх өгөгдөл цэвэрлэгдлээ.');
            }
        }

        function showToast(msg) {
            const toast = document.getElementById('toast');
            document.getElementById('toast-msg').innerText = msg;
            toast.classList.remove('translate-y-20', 'opacity-0');
            setTimeout(() => toast.classList.add('translate-y-20', 'opacity-0'), 3000);
        }
    </script>
